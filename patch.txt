+    def _get_autocomplete_candidates(self, state) -> list[DropdownItem]:
+        """获取自动补全候选词，支持@和/的补全"""
+        text = state.text
+        cursor_position = state.cursor_position
+        
+        # 获取当前单词（从光标位置向前查找）
+        current_word = ""
+        for i in range(cursor_position - 1, -1, -1):
+            if text[i] in [" ", "\n", "\t"]:
+                break
+            current_word = text[i] + current_word
+        
+        candidates = []
+        
+        # @补全：LLM名称（从已加载的LLM配置中获取）

+            llm_names = []

+                agent = self.group_chat.get_members("agent", Agent)
+                if hasattr(agent, 'llm_manager') and agent.llm_manager:
+                    llm_names = list(agent.llm_manager.llms.keys())
+                elif hasattr(agent, 'available_llms'):
+                    llm_names = list(agent.available_llms.keys())
+                else:
+                    # 如果无法从agent获取，尝试从配置文件中获取
+                    from linhai.config import load_config
+                    import os
+                    config_path = os.getenv('LINHAI_CONFIG', 'config.toml')
+                    if os.path.exists(config_path):
+                        config = load_config(config_path)
+                        if "llm" in config:
+                            for llm_config in config["llm"]:
+                                if "name" in llm_config:
+                                    llm_names.append(llm_config["name"])
+            except Exception:
+                # 如果任何步骤失败，返回空列表
+                llm_names = []
+            
+            for llm in llm_names:
